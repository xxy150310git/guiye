服务器状态查询

将以下代码保存为 status.py，然后运行 python status.py 即可查看服务器状态。 

#!/usr/bin/env python3
# 服务器状态查询工具
# 硬编码地址: om.rainplay.cn:14851

import sys
import socket
import json
import struct
import time

SERVER_HOST = "om.rainplay.cn"
SERVER_PORT = 14851

# ---------- Minecraft 协议 ----------
PROTOCOL_VERSION = 754

def _write_varint(buf, value):
    while True:
        if value & ~0x7F == 0:
            buf.append(value)
            return
        buf.append((value & 0x7F) | 0x80)
        value >>= 7

def _read_varint(sock):
    result = 0
    shift = 0
    while True:
        b = sock.recv(1)
        if not b:
            raise EOFError("连接意外关闭")
        byte = b[0]
        result |= (byte & 0x7F) << shift
        if byte & 0x80 == 0:
            return result
        shift += 7

def _read_packet(sock):
    length = _read_varint(sock)
    data = b''
    while len(data) < length:
        chunk = sock.recv(length - len(data))
        if not chunk:
            raise EOFError("连接意外关闭")
        data += chunk
    return data

def query(host, port, timeout=5.0):
    result = {
        "online": False,
        "motd": "",
        "version": "",
        "online_players": 0,
        "max_players": 0,
        "players": [],
        "latency": None,
        "error": None,
    }
    sock = None
    try:
        start_time = time.perf_counter()
        sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
        sock.settimeout(timeout)
        sock.connect((host, port))
        # 握手包
        handshake = bytearray()
        _write_varint(handshake, 0x00)
        _write_varint(handshake, PROTOCOL_VERSION)
        addr_bytes = host.encode('utf-8')
        _write_varint(handshake, len(addr_bytes))
        handshake.extend(addr_bytes)
        handshake.extend(struct.pack('>H', port))
        _write_varint(handshake, 1)
        sock.sendall(_write_varint_and_data(handshake))
        # 状态请求
        request = bytearray()
        _write_varint(request, 0x00)
        sock.sendall(_write_varint_and_data(request))
        # 读取响应
        data = _read_packet(sock)
        pos = 0
        packet_id, pos = _read_varint_from_bytes(data, pos)
        if packet_id != 0x00:
            raise ValueError("无效的响应包 ID")
        json_len, pos = _read_varint_from_bytes(data, pos)
        json_bytes = data[pos:pos + json_len]
        json_str = json_bytes.decode('utf-8')
        status_data = json.loads(json_str)
        result["online"] = True
        desc = status_data.get("description", {})
        if isinstance(desc, dict):
            result["motd"] = desc.get("text", "") or str(desc)
        else:
            result["motd"] = str(desc)
        version_info = status_data.get("version", {})
        result["version"] = version_info.get("name", "")
        players_info = status_data.get("players", {})
        result["online_players"] = players_info.get("online", 0)
        result["max_players"] = players_info.get("max", 0)
        sample = players_info.get("sample", [])
        result["players"] = [p.get("name", "") for p in sample if p.get("name")]
        end_time = time.perf_counter()
        result["latency"] = round((end_time - start_time) * 1000, 2)
    except Exception as e:
        result["error"] = str(e)
    finally:
        if sock:
            sock.close()
    return result

def _write_varint_and_data(data):
    length_prefix = bytearray()
    _write_varint(length_prefix, len(data))
    return bytes(length_prefix) + bytes(data)

def _read_varint_from_bytes(data, offset):
    result = 0
    shift = 0
    i = offset
    while True:
        if i >= len(data):
            raise ValueError("数据不足")
        byte = data[i]
        i += 1
        result |= (byte & 0x7F) << shift
        if byte & 0x80 == 0:
            return result, i
        shift += 7

def main():
    result = query(SERVER_HOST, SERVER_PORT)
    if result.get("error"):
        print("查询失败:", result["error"])
        sys.exit(1)
    print("状态:", "在线" if result["online"] else "离线")
    if result["motd"]:
        print("MOTD:", result["motd"])
    if result["version"]:
        print("版本:", result["version"])
    print("在线人数:", result["online_players"], "/", result["max_players"])
    if result["players"]:
        print("玩家列表:", ", ".join(result["players"]))
    else:
        print("玩家列表: (无)")
    if result["latency"] is not None:
        print("延迟:", result["latency"], "ms")

if __name__ == "__main__":
    main()

复制代码后保存为 status.py，然后在终端运行 python status.py。

如需查询其他服务器，请修改代码开头的 SERVER_HOST 和 SERVER_PORT。

服务器规定

归野 · 生存服务器 生效中 2026-11-21
卡服篇 零容忍
#1
完全禁止卡服行为 禁令

无论出于任何目的，服务器完全禁止任何形式的卡服行为。包括但不限于：

  • 利用游戏漏洞
  • 大量实体堆积
  • 高频红石电路
  • 不当使用加载区块
零容忍政策： 一经发现将视情节严重程度予以 警告、临时封禁 或 永久封禁。
#2
飞行载具高度限制 限高

飞行载具（飞机、飞艇等）飞行高度必须严格控制在 120 格以下，同时 不低于 60 格。

旨在确保空中交通安全有序，避免碰撞，同时优化服务器性能，减少渲染负载。
#3
其他载具运行高度 范围

除飞行载具外，所有载具（地面车辆、船只、矿车等）运行高度必须在 -48 至 256 格范围内。

维护世界结构稳定性与视觉一致性，防止极端高度导致地形加载异常或逻辑错误。
#4
区块常加载 · 需申请 必须审批

如需使用区块常加载功能（自动化农场、刷怪塔、红石装置等），必须 事先填写申请问卷 并获得批准。

https://v.wjx.cn/vm/YYvvb5z.aspx

仅允许弱加载： 必须通过 末影珍珠加载器 实现，严禁使用其他强加载方式。
物品篇 公平
#1
禁止复制「无法量产」物品 禁令

严禁使用物品复制机（含各类红石复制装置、机械动力自动化复制结构等）复制 无法量产 的物品。此类物品在正常游玩中获取数量有限，复制会直接破坏服务器经济与进度平衡。

包括但不限于：

  • 下界之星、龙蛋、鞘翅等唯一性或极稀有物品
  • 附魔金苹果、不死图腾、海洋之心、回响碎片等无法稳定再生的战利品
  • 带有唯一属性或绑定的物品（如限量附魔装备、玩家专属命名物品）
  • 管理组认定的其他不可量产物品
违规处理： 一经查实，复制所得物品将 全部销毁，复制装置 强制拆除，并视情节严重程度予以 警告 至 封禁。
#2
「可量产」的界定 判定

「可量产」指能够通过正常生存玩法 稳定、重复获取 的物品，例如铁、金、钻石、木材、农作物与常规怪物掉落等。是否属于可量产存在争议时，以管理组判定为准，建议提前咨询后再搭建复制设施。

即便复制的是可量产物品，也不得借此进行大规模堆积或造成卡服；相关自动化设施仍须遵守「卡服篇 #4 区块常加载」的申请流程。
铁路篇 机械动力
#1
规划图纸先行 审批

建设铁路前须准备：

  • 详细的铁路线路规划图
  • 途径的所有车站列表
  • 各车站内部设施配置图

全部材料提交给管理员 xxy150310 审核。

完整图纸有助于评估网络合理性、兼容性及服务器性能影响。
#2
枢纽选址须务实 客流导向

车站规划应注重实际客流需求，避免在客流量较小的区域设置大型铁路枢纽。枢纽车站通常需处理多方向、高频率车流，设在客流稀疏处会造成资源浪费并增加管理复杂性。

#3
低客流线路可能搁置 动态调整

若某铁路计划运行区域客流量长期较低，建设申请或已建内容可能被 暂时搁置或作废。鼓励玩家优先在活跃区域发展交通网络。

#4
高度限制 · 隧道建议 工程规范

铁路建设适用载具高度规定：-48 ～ 256 格。隧道可穿越山地，但建议主线 避免在 0 格以下深度 铺设，除非有特殊地形或设计需求。

兼顾工程稳定性与视觉美观。
#5
枢纽必须建设互通连接线 效率

多条线路交汇的车站，必须建设完善的互通连接线，确保不同方向列车高效、安全转换轨道。避免死胡同或单向通行设计。

#6
统一使用「机械动力」模组 强制

服务器内所有铁路及相关设施必须使用 「机械动力」 模组建造。该模组提供丰富的轨道组件、信号系统和动力装置，支持复杂、真实的铁路运营模拟。

地铁篇 机械动力
#1
参照铁路篇审批流程 参照

地铁建设要求与铁路大部分类似，申请时需参照 铁路篇 #1：提交线路图、车站表、设施配置图给 xxy150310 审批。

#2
小客流车站避免大型换乘 务实

小客流车站应尽量避免设计成 三线或以上 的大型换乘站。除非经规划确认为未来区域发展的重要客运枢纽，具有显著客流增长潜力和战略地位。

#3
优先地下 · 地面/高架限高 64 格 视觉管控

地铁应尽量铺设于地下隧道中。若需建设地面站或高架车站，其站体及轨道结构高度 不得大于 64 格。

控制地面视觉影响，保持城市天际线协调，减少对地表生态和其他建筑的遮挡。
#4
统一使用「机械动力」模组 强制

地铁系统必须使用 「机械动力」 模组（含其铁路扩展组件）建造。该模组提供轨道、列车、信号系统与站台设施等专业化组件，可实现地铁与城市轨道交通的高度仿真运营体验。

城镇篇 宜居
#1
高铁车站覆盖要求 通勤

城镇选址或规划时，必须确保其 中心区域 1024 格范围内 设有高铁车站。若高铁站距离超过 512 格，则必须配备有效的通勤连接（如地铁或公交系统）。

确保城镇居民便捷接入服务器主要交通网络。
#2
道路规划 · 照明 安全

城镇内部道路规划需清晰合理。

  • 主干道 宽度不得小于 5 格，保证双向通行及紧急疏散。
  • 所有道路（尤其是主干道和居民区街道）需配备 充足路灯或照明设施，保证夜间安全。
营造宜居、安全的夜间环境。
#3
建筑高度与景观协调 美学

建筑高度无严格上限，但应充分考虑与周边自然景观、既有建筑群的协调性。避免突兀、比例失调的巨型结构，提倡 多样化和有层次的建筑风格，形成和谐统一的视觉印象。

#4
禁止大型工业设施 禁止

为维护居住品质，城镇区域内 严格禁止 设立大型工业设施（大型工厂、重型加工场等）。这些设施可能产生持续噪音、视觉污染或影响游戏性能。

营造宁静、舒适的居住体验。
#5
公共区域整洁有序 强制

所有公共区域（广场、公园、街道、社区中心）必须保持整洁。

  • 严禁随意丢弃物品、堆放垃圾
  • 严禁私自搭建临时性或违章建筑
违规处理： 收到整改通知后若未在规定时间内处理，相关物品或建筑将 强制拆除，玩家可能受到处罚。
